cmake_minimum_required(VERSION 3.21)

# bwapi-c2: a flat C ABI over BWAPI and BWEM. docs/c-abi-plan.md is the design; section 7 is
# the layout, section 10.1 the closure this file builds, section 11 the test suite.
project(bwapi_c2 VERSION 0.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# The DLL statically links the CRT (/MT): no allocation crosses the boundary (section 4), and
# section 0 permits it. CMP0091 is NEW at this minimum version, so this variable is honoured.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# The documented hook, carried over from upstream's CMake (section 10.1): extra flags for the
# BWAPI/BWEM closure only.
set(BWAPI_CUSTOM_COMPILE_FLAGS "" CACHE STRING "Extra compile options for the BWAPI/BWEM closure")

option(BWAPI_C2_BUILD_TESTS "Build the test suite (section 11)" ${PROJECT_IS_TOP_LEVEL})

# BWEM's own assertions (section 8.3, R11.9). Upstream compiles bwem_assert to nothing unless
# BWEM_ASSERTS is defined, and BWEM accepts a partially overlapping neutral silently in that
# state. With this ON, a fixture mistake fails at the point of cause instead of at teardown. Pair
# it with a Debug build: assert() is itself a no-op under NDEBUG. Non-MSVC only; see closure.cmake.
option(BWAPI_C2_BWEM_ASSERTS "Compile BWEM with its assertions enabled (non-MSVC, Debug build)" OFF)

# Sanitizers for the whole tree, closure included: "address,undefined" is what CI's second Linux
# configuration runs the suite under (implementation plan 0.11). Empty means none.
set(BWAPI_C2_SANITIZERS "" CACHE STRING "Comma-separated -fsanitize= list applied to every target")
if(BWAPI_C2_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=${BWAPI_C2_SANITIZERS} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${BWAPI_C2_SANITIZERS})
endif()

# Both dependencies are submodules pinned to our forks (docs/pins.md). Say what to do when they
# are missing, and say it here rather than as a compile error five minutes later.
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/bwapi/bwapi/include/BWAPI.h"
   OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/bwem/BWEM/include/bwem.h")
  message(FATAL_ERROR
    "The submodules under third_party/ are not checked out. Run\n"
    "    git submodule update --init --depth 1\n"
    "without --recursive: BWEM's own submodules must never be fetched (docs/pins.md).")
endif()

include(cmake/closure.cmake)

# ---------------------------------------------------------------------------------------------
# The ABI's own sources (section 7): the hand-written boundary, lifecycle and handles, and from
# phase 1 on the generated *.gen.cpp. An object library rather than part of the shared library
# directly, because the test suite links these same objects beside the fixture's copy of the
# closure: one BroodwarPtr, one error latch, and every export exercised in-process without a
# second copy of BWAPI. Every file is named; never glob.
add_library(bwapi_c2_abi OBJECT
  src/abi.cpp
  src/client.cpp
  src/handles.cpp
  src/bulk.cpp
  # Generated by tools/abi/emit_source.py, one per spec file, and checked in (implementation
  # plan rule 2). emit_source.py refuses to write a file that is not named here.
  src/bulk.gen.cpp
  src/client.gen.cpp
  src/constants.gen.cpp
  src/player.gen.cpp
  src/types.gen.cpp
)
target_link_libraries(bwapi_c2_abi PUBLIC bwapi_c2_closure)
target_include_directories(bwapi_c2_abi
  PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
  PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src"
)
# PUBLIC: a test that links these objects on Windows must see the same dllexport declarations
# the objects were compiled against, or MSVC looks for __imp_ symbols that do not exist.
target_compile_definitions(bwapi_c2_abi
  PUBLIC BWAPI_C2_BUILDING=1
  PRIVATE
    BWAPI_C2_VERSION_MAJOR=${PROJECT_VERSION_MAJOR}
    BWAPI_C2_VERSION_MINOR=${PROJECT_VERSION_MINOR}
    BWAPI_C2_VERSION_PATCH=${PROJECT_VERSION_PATCH}
)
set_target_properties(bwapi_c2_abi PROPERTIES
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
  POSITION_INDEPENDENT_CODE ON
)
if(MSVC)
  target_compile_options(bwapi_c2_abi PRIVATE /W4)
else()
  target_compile_options(bwapi_c2_abi PRIVATE -Wall -Wextra -pedantic)
endif()

# The shared library: the ABI objects plus the closure. Named bwapi_c2.dll / libbwapi_c2.so;
# exports are listed by name in bwapi_c2.def (section 4) and CI diffs dumpbin's view against it.
# Object libraries do not propagate their objects transitively, so both are linked here.
add_library(BWAPI_C2 SHARED)
set_target_properties(BWAPI_C2 PROPERTIES
  OUTPUT_NAME bwapi_c2
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
)
target_link_libraries(BWAPI_C2 PRIVATE bwapi_c2_abi bwapi_c2_closure)
target_include_directories(BWAPI_C2 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
  # CMake passes a .def listed among the sources to the linker as /DEF.
  target_sources(BWAPI_C2 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/bwapi_c2.def")
endif()

if(BWAPI_C2_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
